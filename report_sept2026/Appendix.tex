\section{Implementation Details}
Table~\ref{tab:config} lists the training configuration of the group word-problem experiments (Section~\ref{sec:setup}). Table~\ref{tab:precision} verifies the choice of float32 for the halting nonlinearity. A gate whose logits were set to 10, beyond the bfloat16 threshold, was optimised for 300 steps on the budget penalty alone. In bfloat16 it receives a gradient of exactly zero and never moves; in float32 it closes from 3.9995 to 0.0625 factors. With small logits both precisions behave identically, so the effect is specific to saturation. The gate initialisation was verified in the same code path used for training: with the model class's default initialisation the initial effective order was 0.9376, whereas with explicit initialisation it was 3.8233 for every token, and every run in Table~\ref{tab:sweep} logged 3.823 at its first step. Algorithm~\ref{alg:forward} gives the forward pass of the adaptive-order layer and Algorithm~\ref{alg:train} the training procedure, and Table~\ref{tab:env} records the environment.

\begin{algorithm}[ht]
\caption{Forward pass of the adaptive-order layer for one token}
\label{alg:forward}
\KwIn{token representation $h_t$; previous state $S_{t-1}$; maximum order $K$; mode (training or evaluation)}
\KwOut{new state $S_t$; effective order $n_t$}
$z_t \leftarrow W_2\,\mathrm{GELU}(W_1 h_t + b_1) + b_2$ \tcp*{$K$ gate logits}
$\lambda_{t,i} \leftarrow \sigma(z_{t,i})$ for $i = 1,\dots,K$ \tcp*{evaluated in float32}
$g_{t,i} \leftarrow \prod_{j=1}^{i}\lambda_{t,j}$ \tcp*{monotone gates}
\If{mode = evaluation}{
  $g_{t,i} \leftarrow \mathbf{1}\left[g_{t,i} > 0.5\right]$ \tcp*{hard gates, integer order}
}
$n_t \leftarrow \sum_{i=1}^{K} g_{t,i}$\;
$S \leftarrow S_{t-1}$\;
\For{$i \leftarrow 1$ \KwTo $K$}{
  $\tilde\beta_{t,i} \leftarrow g_{t,i}\cdot 2\,\sigma(w_i^\top h_t)$\;
  $S \leftarrow S\,(I - \tilde\beta_{t,i}\,k_{t,i}k_{t,i}^\top) + \tilde\beta_{t,i}\, v_{t,i}k_{t,i}^\top$\;
}
\Return{$S_t \leftarrow S$, \ $n_t$}\;
\end{algorithm}

\begin{algorithm}[ht]
\caption{Training the adaptive-order layer with a compute budget}
\label{alg:train}
\KwIn{training sequences; penalty weight $\gamma$; warm-up $s_w$; ramp $s_r$; number of steps $N$}
\KwOut{trained model}
Initialise the gate output layer with zero weights and bias $b_0$\;
\For{$s \leftarrow 1$ \KwTo $N$}{
  Sample a batch of $B$ sequences of length $T$\;
  Run Algorithm~\ref{alg:forward} in training mode over every position\;
  $\mathcal{L}_{\text{task}} \leftarrow$ cross-entropy of the readout against $y_{b,t}$\;
  $\mathcal{L} \leftarrow \mathcal{L}_{\text{task}} + \gamma(s)\cdot\frac{1}{BT}\sum_{b,t} n_{b,t}$ \tcp*{Equation \ref{eq:loss}}
  Update all parameters with AdamW\;
}
Evaluate with hard gates on the held-out sequences\;
\end{algorithm}
